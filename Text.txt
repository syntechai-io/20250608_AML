        function generateQuiz() {
            const container = document.getElementById('quizContainer');
            container.innerHTML = '';
            
            quizQuestions.forEach((q, index) => {
                const questionDiv = document.createElement('div');
                questionDiv.className = 'quiz-question';
                questionDiv.innerHTML = `
                    <h3>Frage ${index + 1}</h3>
                    <p>${q.question}</p>
                    <div class="quiz-options">
                        ${q.options.map((option, optIndex) => `
                            <label class="quiz-option">
                                <input type="radio" name="question${index}" value="${optIndex}">
                                ${option}
                            </label>
                        `).join('')}
                    </div>
                `;
                container.appendChild(questionDiv);
            });
            
            document.getElementById('submitQuiz').style.display = 'inline-block';
        }

        function submitQuiz() {
            quizAnswers = {};
            quizScore = 0;
            
            quizQuestions.forEach((q, index) => {
                const selected = document.querySelector(`input[name="question${index}"]:checked`);
                if (selected) {
                    const answer = parseInt(selected.value);
                    quizAnswers[index] = answer;
                    if (answer === q.correct) {
                        quizScore++;
                    }
                }
            });
            
            if (Object.keys(quizAnswers).length < quizQuestions.length) {
                alert('Bitte beantworten Sie alle Fragen.');
                return;
            }
            
            if (quizScore === quizQuestions.length) {
                userData.completionTime = new Date().toISOString();
                userData.quizScore = quizScore;
                showCertificate();
            } else {
                alert(`Sie haben ${quizScore} von ${quizQuestions.length} Fragen richtig beantwortet. Sie benötigen 100% (${quizQuestions.length}/${quizQuestions.length}) zum Bestehen.`);
                document.getElementById('retakeQuiz').style.display = 'inline-block';
            }
        }

        function retakeQuiz() {
            document.getElementById('retakeQuiz').style.display = 'none';
            generateQuiz();
        }

        function showCertificate() {
            document.querySelectorAll('.section').forEach(section => {
                section.classList.remove('active');
            });
            document.getElementById('certificate').classList.add('active');
            
            document.getElementById('certificateName').textContent = `${userData.firstName} ${userData.lastName}`;
            document.getElementById('certificateDate').textContent = new Date().toLocaleDateString('de-DE');
            document.getElementById('certificateId').textContent = userData.certificateId;
            
            updateProgress();
        }

        function generateCertificateId() {
            return 'CASP-' + Date.now().toString(36).toUpperCase() + '-' + Math.random().toString(36).substr(2, 4).toUpperCase();
        }

        function updateProgress() {
            const totalSteps = 7; // Registration + 5 modules + quiz
            let currentStep = currentModule;
            if (document.getElementById('quiz').classList.contains('active')) currentStep = 6;
            if (document.getElementById('certificate').classList.contains('active')) currentStep = 7;
            
            const progress = (currentStep / totalSteps) * 100;
            document.getElementById('progressBar').style.width = progress + '%';
        }

        function downloadData() {
            const csvData = [
                ['Vorname', 'Nachname', 'Email', 'Firma', 'Startzeit', 'Abschlusszeit', 'Quiz-Ergebnis', 'Zertifikat-ID'],
                [userData.firstName, userData.lastName, userData.email, userData.company, userData.startTime, userData.completionTime, userData.quizScore + '/8', userData.certificateId]
            ];
            
            const csvContent = csvData.map(row => row.join(',')).join('\n');
            const blob = new Blob([csvContent], { type: 'text/csv' });
            const url = window.URL.createObjectURL(blob);
            
            const a = document.createElement('a');
            a.href = url;
            a.download = `AML_CFT_Training_${userData.certificateId}.csv`;
            a.click();
            
            window.URL.revokeObjectURL(url);
        }

        function printCertificate() {
            window.print();
        }

        // Initialize
        updateProgress();
